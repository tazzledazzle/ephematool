package auth additions:
  const RoleAdmin = "admin"; RoleCreator = "creator"; RoleViewer = "viewer"
  type TeamChecker interface{ IsTeamMember(ctx context.Context, org, slug, login string) (bool, error) }   // *ghapp.Client satisfies it
  type RoleResolver struct{ AdminLogins, CreatorLogins, CreatorTeams []string; Teams TeamChecker }
  func NewRoleResolver(cfg config.Config, teams TeamChecker) RoleResolver
  func (r RoleResolver) Resolve(ctx context.Context, login string) (string, error)
      // order: AdminLogins (EqualFold) -> CreatorLogins (EqualFold) -> each CreatorTeams "org/slug" via Teams (nil Teams => skip, log once auth.creator_teams_unchecked) -> viewer.
      // A team-check error yields viewer plus the error (fail closed); callers log auth.role_check_failed.
  func RoleAtLeast(role, min string) bool        // viewer < creator < admin
  type Principal struct{ GithubID int64; Login, AvatarURL string }
  func PrincipalFrom(ctx context.Context, sm *scs.SessionManager) (Principal, bool)
  func RequireRole(api huma.API, sm *scs.SessionManager, r RoleResolver, min string) func(huma.Context, func(huma.Context))
      // per-operation middleware (huma.Operation.Middlewares); 403 "access denied" when below min or on resolve error
  const PublicExtension = "x-public"; func MarkPublic(op *huma.Operation)   // sets op.Extensions["x-public"] = true
  Handlers gains field Roles RoleResolver; login-time hint uses an adapter over the user-token MembershipClient.TeamMembership
